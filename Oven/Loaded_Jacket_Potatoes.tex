\begin{recipebox}{Loaded Jacket Potatoes}{2--3}{60 min}
\begin{ingredients}
  \item 3 large potatoes
  \item 1 cup chilli beans or baked beans
  \item ½ cup grated cheese
  \item ¼ cup light sour cream or yoghurt
  \item Spring onion, salad on the side
\end{ingredients}
\begin{method}
  \item Bake potatoes at 200°C for 50--60 min (or microwave then finish in oven).
  \item Split, fluff insides; top with hot beans and cheese.
  \item Finish with sour cream/yoghurt and spring onion.
\end{method}
\textbf{Source:} \href{https://www.bbcgoodfood.com/recipes/jacket-potatoes}{BBC Good Food – Jacket Potatoes}
\end{recipebox}
