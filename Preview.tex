% --- Pretty recipe boxes ---
\usepackage[most]{tcolorbox}
\tcbset{
  recipe style/.style={
    enhanced, breakable,
    colback=white,
    colframe=black!20,
    boxrule=0.6pt,
    arc=3mm,
    drop shadow,
    left=6mm,right=6mm,top=4mm,bottom=4mm,
  }
}
\newenvironment{recipebox}[3]% #1 Title, #2 Servings, #3 Time
{%
  \phantomsection
  \addcontentsline{toc}{section}{#1}% add recipe to chapter mini-TOC
  \begin{tcolorbox}[recipe style, title={\textbf{\large #1}}]
  \textit{Servings:} #2 \hfill \textit{Time:} #3
  \par\medskip
}
{\end{tcolorbox}}

% Sub-environments for structure inside the box
\newenvironment{ingredients}
  {\subsection*{Ingredients}\begin{itemize}[leftmargin=2em]}
  {\end{itemize}\medskip}

\newenvironment{method}
  {\subsection*{Method}\begin{enumerate}[leftmargin=2em]}
  {\end{enumerate}}

% Sub-environment for optional notes (paragraph style)
\newenvironment{notes}
  {\subsection*{Notes}\begin{quote}\small\itshape}
  {\end{quote}\medskip}

% (Optional) Bullet-style notes if you ever want them
\newenvironment{notelist}
  {\subsection*{Notes}\begin{itemize}[leftmargin=2em]}
  {\end{itemize}\medskip}
