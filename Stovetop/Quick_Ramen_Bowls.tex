\begin{recipebox}{Quick Ramen Bowls}{2--3}{25 min}
\begin{ingredients}
  \item 1.25 L light chicken or miso broth
  \item 2 packs ramen or egg noodles
  \item 2 eggs (soft-boiled)
  \item 1 cup mixed vegetables (bok choy, corn, mushrooms)
  \item 200 g sliced cooked chicken or tofu
  \item 2 tbsp soy sauce; optional: sesame oil, chilli
\end{ingredients}
\begin{method}
  \item Simmer broth with soy (and sesame/chilli if using).
  \item Add veg; cook 3--4 min. Add noodles; cook until tender.
  \item Bowl it up; top with chicken/tofu and halved eggs.
\end{method}
\textbf{Source:} \href{https://www.bbcgoodfood.com/recipes/chicken-ramen}{BBC Good Food – Chicken Ramen}
\end{recipebox}
