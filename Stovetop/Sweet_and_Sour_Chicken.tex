\begin{recipebox}{Week 12 – Sweet \& Sour Chicken}{2--3}{35 min}
\begin{ingredients}
  \item 300 g chicken breast, diced
  \item 1 red capsicum, chopped
  \item 1 carrot, sliced
  \item 1 small can pineapple chunks
  \item 2 tbsp sweet \& sour sauce
  \item 2 cups rice
\end{ingredients}

\begin{method}
  \item Cook chicken in pan until browned.
  \item Add vegetables, pineapple, and sauce. Simmer 10 min.
  \item Serve with steamed rice.
\end{method}
\textbf{Source:} \href{https://www.bbcgoodfood.com/recipes/sweet-sour-chicken}{BBC Good Food – Sweet \& Sour Chicken}
\end{recipebox}
