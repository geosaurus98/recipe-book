\begin{recipebox}{Homemade Sub Sandwiches}{2--3}{15–20 min}
\begin{ingredients}
  \item 3 crusty rolls or sub buns
  \item 200 g deli meat (chicken, ham, turkey) or roasted veg
  \item Slices of cheese
  \item Lettuce, tomato, cucumber, red onion, pickles
  \item Mustard, light mayo or yoghurt-garlic sauce
\end{ingredients}
\begin{method}
  \item Split rolls; layer meat/veg, cheese, and salad.
  \item Add sauces; press and serve (toast if you like).
\end{method}
\textbf{Source:} \href{https://www.taste.com.au/recipes/subway-style-subs/}{Taste – Subway-Style Subs}
\end{recipebox}
