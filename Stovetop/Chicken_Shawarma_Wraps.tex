\begin{recipebox}{Chicken Shawarma Wraps}{2--3}{30–35 min}
\begin{ingredients}
  \item 300\,g chicken thighs, shawarma-spiced
  \item 4 flatbreads or wraps
  \item Lettuce, tomato, red onion, pickles
  \item Garlic yoghurt sauce or toum
\end{ingredients}
\begin{method}
  \item Pan-fry or oven-bake spiced chicken; slice.
  \item Warm flatbreads; add salad, chicken, and sauce.
  \item Wrap and serve (with oven fries if desired).
\end{method}
\textbf{Source:} \href{https://www.recipetineats.com/chicken-shawarma-middle-eastern/}{RecipeTin Eats – Chicken Shawarma}
\end{recipebox}
