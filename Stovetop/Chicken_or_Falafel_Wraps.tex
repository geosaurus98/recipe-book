\begin{recipebox}{Chicken or Falafel Wraps}{2--3}{30 min}
\begin{ingredients}
  \item 300\,g chicken breast or falafel balls
  \item 4 wraps
  \item Hummus
  \item Salad mix (lettuce, cucumber, tomato)
  \item Garlic yoghurt sauce
  \item Oven fries or tabbouleh
\end{ingredients}

\begin{method}
  \item Cook chicken or heat falafel.
  \item Spread hummus on wraps, add salad and protein.
  \item Drizzle with yoghurt sauce, wrap, and serve with side.
\end{method}
\textbf{Source:} \href{https://www.bbcgoodfood.com/recipes/chicken-kebabs}{BBC Good Food – Chicken Kebabs}
\end{recipebox}
