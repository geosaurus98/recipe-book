\begin{recipebox}{Burrito Bowls}{2--3}{30 min}
\begin{ingredients}
  \item 2 cups cooked rice (brown or white)
  \item 1 can black beans, rinsed
  \item 300 g cooked chicken or beef (or roasted veg)
  \item 1 cup corn kernels
  \item 1 tomato, diced; 1 avocado, sliced
  \item ½ cup grated cheese; salsa, lime, coriander
\end{ingredients}
\begin{method}
  \item Warm beans, corn, and protein in a pan.
  \item Assemble bowls: rice base, then toppings.
  \item Finish with salsa, lime juice, and coriander.
\end{method}
\textbf{Source:} \href{https://www.bbcgoodfood.com/recipes/chicken-burrito-bowl}{BBC Good Food – Chicken Burrito Bowl}
\end{recipebox}
