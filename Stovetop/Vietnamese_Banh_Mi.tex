\begin{recipebox}{Vietnamese Banh Mi}{2--3}{35 min (+ quick pickle)}
\begin{ingredients}
  \item 3 small crusty baguettes
  \item 300\,g chicken/pork (or tofu), marinated and pan-seared
  \item Quick pickle: carrot \& cucumber with a splash of vinegar \& sugar
  \item Fresh coriander, sliced chilli (optional)
  \item Light mayo; soy or Maggi seasoning
\end{ingredients}
\begin{method}
  \item Make quick pickle; set aside 10--15 min.
  \item Cook marinated protein; slice.
  \item Split baguettes; spread mayo, layer protein, pickles, herbs, chilli. Splash with seasoning.
\end{method}
\textbf{Source:} \href{https://www.recipetineats.com/banh-mi-vietnamese-sandwich/}{RecipeTin Eats – Banh Mi}
\end{recipebox}
