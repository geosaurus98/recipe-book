\begin{recipebox}{Chicken Tikka Masala}{2--3}{35 min}
\begin{ingredients}
  \item 300 g chicken breast, diced
  \item 1 onion, chopped
  \item 1 tbsp tikka masala paste
  \item 1 can (400 g) chopped tomatoes
  \item \nicefrac{1}{2} cup plain yoghurt (or coconut milk)
  \item 2 cups rice
  \item Optional naan bread
\end{ingredients}

\begin{method}
  \item Cook onion until soft, stir in curry paste.
  \item Add chicken, cook 5 min.
  \item Stir in tomatoes, simmer 15 min.
  \item Add yoghurt, cook 2–3 min.
  \item Serve with rice and naan.
\end{method}
\textbf{Source:} \href{https://www.bbcgoodfood.com/recipes/chicken-tikka-masala}{BBC Good Food – Chicken Tikka Masala}
\end{recipebox}
