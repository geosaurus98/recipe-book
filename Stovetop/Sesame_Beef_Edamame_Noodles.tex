\begin{recipebox}{Sesame Beef, Edamame \& Vegetable Noodles}{4}{30 min}

\begin{ingredients}
  \item 500\,g lean beef mince
  \item 250\,g dried egg noodles
  \item 300\,g frozen shelled edamame
  \item 700\,g frozen stir-fry vegetables
  \item 1 onion, thinly sliced
  \item 3 garlic cloves, crushed
  \item 2 tsp grated ginger
  \item 4 tbsp reduced-salt soy sauce
  \item 1 tbsp sweet chilli sauce or honey
  \item 1 tbsp rice vinegar or lemon juice
  \item 1 tsp sesame oil
  \item \nicefrac{1}{2} cup water
  \item Chilli flakes, to taste (optional)
\end{ingredients}

\begin{method}
  \item Cook the noodles according to the packet instructions, then drain and set aside.
  \item Brown the mince in a large wok or deep frying pan over high heat for 5--6 min, breaking it up as it cooks. Drain excess fat if necessary.
  \item Add the onion, garlic and ginger and cook for a further 2 min.
  \item Add the frozen vegetables, edamame and water. Stir-fry for 5--6 min, until the water has cooked off and the vegetables are hot through.
  \item Stir in the soy sauce, sweet chilli sauce and vinegar.
  \item Toss through the noodles, finish with the sesame oil, and divide into four portions.
\end{method}

\begin{notelist}
  \item Roughly 40\,g protein per serve.
  \item Defrost the mince overnight if frozen. The edamame and vegetables can go in straight from frozen.
  \item Egg noodles are the cheap default; udon or rice noodles also work.
  \item Extra veg is easy to add: a handful of shredded cabbage or grated carrot goes in with the frozen vegetables.
\end{notelist}

\end{recipebox}
