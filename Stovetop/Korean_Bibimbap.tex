\begin{recipebox}{Korean Bibimbap}{2--3}{35 min}
\begin{ingredients}
  \item 2 cups cooked rice
  \item 200 g sliced beef or chicken (or tofu)
  \item 2 cups assorted veg (spinach, carrot, mushrooms, courgette)
  \item 2 eggs
  \item 2 tbsp soy sauce, 1 tsp sugar, 1 tsp sesame oil
  \item Gochujang (Korean chilli paste) to serve
\end{ingredients}
\begin{method}
  \item Stir-fry protein; season with soy, sugar, sesame. Set aside.
  \item Stir-fry veg separately with a little oil; season lightly.
  \item Fry eggs sunny-side up.
  \item Serve rice topped with protein, veg, egg, and gochujang.
\end{method}
\textbf{Source:} \href{https://www.bbcgoodfood.com/recipes/bibimbap}{BBC Good Food – Bibimbap}
\end{recipebox}
