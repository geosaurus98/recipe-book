\begin{recipebox}{Instant Pot Pasta with Sausage and Spinach}{6}{35 min}
  \begin{ingredients}
    \item 2 teaspoons olive oil
    \item 1 lb Italian sausage (casings removed)
    \item 1 medium onion, diced
    \item 3 cloves garlic, minced
    \item 1 (24 oz) jar marinara sauce
    \item 3 cups chicken broth
    \item 12 oz uncooked penne pasta (or similar)
    \item \nicefrac{1}{2} teaspoon salt (adjust if broth is salted)
    \item \nicefrac{1}{4} teaspoon black pepper
    \item 4 cups fresh baby spinach
    \item \nicefrac{1}{2} cup grated Parmesan cheese (for serving)
  \end{ingredients}

  \begin{method}
    \item Set Instant Pot to sauté mode. Add olive oil and sausage; cook until browned, breaking up with a spoon.
    \item Add onion and garlic; cook a few minutes until softened.
    \item Stir in marinara sauce, chicken broth, pasta, salt, and pepper. Mix well to submerge pasta.
    \item Cancel sauté. Secure lid, set valve to sealing. Pressure cook on HIGH for 5 minutes.
    \item Quick release steam. Stir in fresh spinach until wilted.
    \item Serve immediately, topped with Parmesan cheese.
  \end{method}
\end{recipebox}
