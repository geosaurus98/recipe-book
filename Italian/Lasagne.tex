\begin{recipebox}{Classic Bolognese Lasagne}{6--8}{2 hr 30 min}

  \begin{ingredients}
    \begin{multicols}{2}
      \item \textbf{For the meat sauce (ragu):}
      \item 600\,g mince (beef and pork, 2:1 ratio)
      \item 1 carrot, finely chopped
      \item 1 celery stalk, finely chopped
      \item 1 onion, finely chopped
      \item 700\,ml tomato purée (passata)
      \item 2 tbsp tomato paste
      \item \nicefrac{1}{2} cup red wine
      \item Extra virgin olive oil
      \item Salt and pepper, to taste
      \columnbreak
      \item \textbf{For the béchamel sauce:}
      \item 1\,L (4 cups) whole milk (or lactose-free milk)
      \item 100\,g (7 tbsp) butter (or dairy-free margarine)
      \item 100\,g (\nicefrac{3}{4} cup) plain flour
      \item Nutmeg, to taste
      \item Salt, to taste
      \vspace{0.4em}
      \item \textbf{Other ingredients:}
      \item 250--300\,g fresh or dry egg lasagne sheets
      \item 100\,g grated parmesan cheese (or lactose-free tasty cheese)
    \end{multicols}
  \end{ingredients}

  \vspace{-1em}
  \begin{method}
    \item \textbf{Make the Ragu:}
    \begin{itemize}[leftmargin=2em]
      \item Finely chop the onion, carrot, and celery.
      \item Sauté the vegetables in a drizzle of olive oil until soft.
      \item Add the ground meat and cook until browned.
      \item Pour in the red wine and let it evaporate.
      \item Add the tomato purée and paste, then season with salt and pepper.
      \item Simmer covered on low heat for 1.5--2 hours, stirring occasionally. The longer it cooks, the richer the flavour.
    \end{itemize}

    \vspace{0.3em}
    \item \textbf{Make the Béchamel Sauce:}
    \begin{itemize}[leftmargin=2em]
      \item Melt the butter in a saucepan.
      \item Add the flour and stir to create a smooth paste (roux).
      \item Gradually whisk in the milk to avoid lumps.
      \item Cook over medium heat until thickened.
      \item Season with salt and a pinch of nutmeg.
    \end{itemize}

    \vspace{0.3em}
    \item \textbf{Assemble the Lasagne:}
    \begin{itemize}[leftmargin=2em]
      \item Spread a little béchamel and ragu on the bottom of a baking dish.
      \item Add a layer of lasagne sheets, followed by ragu, béchamel, and a sprinkle of parmesan.
      \item Continue layering until all ingredients are used, ending with béchamel and a generous layer of parmesan on top.
    \end{itemize}

    \vspace{0.3em}
    \item \textbf{Bake and Rest:}
    \begin{itemize}[leftmargin=2em]
      \item Bake for 40--45 minutes at 180\degree C.
      \item For a golden top, turn on the grill or broiler for the final 5 minutes.
      \item Allow the lasagne to rest for 15--20 minutes before serving. This helps it set and enhances the flavour.
    \end{itemize}

  \end{method}

  \begin{notes}
    Pairs beautifully with a glass of full-bodied red wine.
  \end{notes}

\end{recipebox}
