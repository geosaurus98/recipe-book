\begin{recipebox}{Rigatoni all’Amatriciana}{4}{45 min}

  \begin{ingredients}
    \item 500\,g Rigatoni (La Molisana brand)
    \item 200\,g guanciale or streaky bacon
    \item 1 small onion (red or white)
    \item 2 cans whole tomatoes or 800\,g passata
    \item \nicefrac{1}{2} glass white wine
    \item Extra virgin olive oil
    \item 100\,g pecorino cheese
    \item Salt and pepper, to taste
  \end{ingredients}

  \vspace{-0.5em}
  \begin{method}

    \item \textbf{Step 1 — Prepare the Onion Base:}
    \begin{itemize}[leftmargin=2em]
      \item In a little olive oil over low heat, fry the onion until soft and translucent.
    \end{itemize}

    \vspace{0.5em}

    \item \textbf{Step 2 — Cook the Guanciale:}
    \begin{itemize}[leftmargin=2em]
      \item Cut the guanciale or streaky bacon into cubes or strips.
      \item Fry in a bit of oil until golden and slightly crisp.
    \end{itemize}

    \vspace{0.5em}

    \item \textbf{Step 3 — Deglaze with White Wine:}
    \begin{itemize}[leftmargin=2em]
      \item Raise the heat and add the white wine.
      \item Let the alcohol evaporate completely for a couple of minutes.
    \end{itemize}

    \vspace{0.5em}

    \item \textbf{Step 4 — Add the Tomatoes:}
    \begin{itemize}[leftmargin=2em]
      \item Lower the heat and add the tomatoes or passata.
      \item Season lightly with salt and pepper — remember that pecorino will add saltiness later.
      \item Bring to a gentle boil.
    \end{itemize}

    \vspace{0.5em}

    \item \textbf{Step 5 — Simmer the Sauce:}
    \begin{itemize}[leftmargin=2em]
      \item Reduce to medium-low heat and let the sauce simmer for about 30 minutes.
      \item Stir occasionally until the sauce thickens and flavours deepen.
    \end{itemize}

    \vspace{0.5em}

    \item \textbf{Step 6 — Cook the Pasta:}
    \begin{itemize}[leftmargin=2em]
      \item Boil the Rigatoni in salted water until al dente.
      \item Drain well, reserving a small amount of the pasta water.
    \end{itemize}

    \vspace{0.5em}

    \item \textbf{Step 7 — Combine and Serve:}
    \begin{itemize}[leftmargin=2em]
      \item Add the cooked pasta to the sauce pan and toss to coat evenly.
      \item Mix in grated pecorino cheese, adding a splash of pasta water if needed to loosen the sauce.
      \item Serve immediately with extra pecorino on top.
    \end{itemize}

  \end{method}

  \begin{notes}
    When combining pasta and sauce, use a large bowl: spread half the sauce on the bottom, add the pasta, mix well, then add the remaining sauce and mix again before serving.
  \end{notes}

\end{recipebox}
