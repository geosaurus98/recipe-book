\begin{recipebox}{Rustic Chickpea \& Rocket Salad with Mustard-Garlic Vinaigrette}{2}{15 min}

  \begin{ingredients}
    \item \textbf{For the salad:}
    \item 100\,g rocket or mixed greens
    \item 200\,g cooked chickpeas (canned is fine, drained and rinsed)
    \item 10 cherry tomatoes, halved
    \item \nicefrac{1}{4} red onion, thinly sliced
    \item 2 tbsp pitted olives
    \item Parmesan or pecorino shavings (optional)
    \vspace{0.6em}
    \item \textbf{For the dressing:}
    \item 1 garlic clove, finely minced or crushed
    \item \nicefrac{1}{2} tsp salt
    \item Black pepper, to taste
    \item 3 tbsp extra virgin olive oil
    \item 1 tsp Dijon or mild mustard
    \item 1 tbsp white wine vinegar or apple cider vinegar
  \end{ingredients}

  \vspace{-0.5em}
  \begin{method}

    \item \textbf{Step 1 — Make the Dressing:}
    \begin{itemize}[leftmargin=2em]
      \item In a small bowl, combine the garlic, salt, and pepper.
      \item Add the mustard and vinegar, and mix well.
      \item Slowly drizzle in the olive oil while whisking to emulsify.
    \end{itemize}

    \vspace{0.8em}

    \item \textbf{Step 2 — Prepare the Salad:}
    \begin{itemize}[leftmargin=2em]
      \item In a large bowl, add the rocket, chickpeas, tomatoes, red onion, and olives.
      \item Add cheese shavings if using.
    \end{itemize}

    \vspace{0.8em}

    \item \textbf{Step 3 — Dress and Serve:}
    \begin{itemize}[leftmargin=2em]
      \item Pour the dressing over the salad just before serving.
      \item Toss gently to coat everything evenly.
      \item Serve immediately — optionally with toasted bread on the side.
    \end{itemize}

  \end{method}

  \begin{notes}
    Light, fresh, and protein-rich — perfect as a summer lunch or a side to grilled meat.
  \end{notes}

\end{recipebox}
