\begin{recipebox}{DIY Pizzas}{2--3}{20 min}
\begin{ingredients}
  \item 4 tsp Surebake yeast
  \item 450\,g flour
  \item 1 tsp Salt
  \item 1 tsp Sugar
  \item 5 tbsp Olive Oil
  \item 230\,ml Water
  \item \nicefrac{1}{2} cup pizza sauce or passata
  \item 1 cup shredded mozzarella (or lactose-free mozzarella)
  \item Toppings: mushrooms, capsicum, olives, pineapple, chicken, ham
\end{ingredients}

\begin{method}
  \item Put Top 6 items into breadmaker on Pizza setting for 45 minutes
  \item Knead dough on a lightly floured surface until it becomes elastic and springs back when touched.
  \item Divide into 2 large balls and place in a warm area for 10 minutes
  \item Shape each ball into a flat circle. Place the circles on a baking tray and prick with a fork
  \item Leave to rise for a further 15 minutes
  \item Preheat oven to 220°C.
  \item Spread sauce, add cheese and toppings.
  \item Bake 8–10 min for pita bread or 15-20 min for pizza bread until golden. Slice and serve.
\end{method}
\begin{notes}
Topping options could include mushrooms, capsicum, olives, pineapple, chicken, ham, Salami, tomatoes, red onion, spinach, corn, etc
\end{notes}
\end{recipebox}
