\begin{recipebox}{Butter Chicken}{4}{40 min}
\begin{ingredients}
  \item 400 g chicken thighs, boneless and skinless, diced
  \item 1 onion, finely chopped
  \item 2 tsp garam masala
  \item 2 tsp cumin
  \item 2 tsp paprika
  \item 1 tsp turmeric
  \item 1 tsp ground coriander
  \item 2 cloves garlic, minced
  \item 1 Tbsp grated ginger
  \item 400 g can crushed tomatoes (or tomato purée)
  \item ½ cup cream (or coconut cream)
  \item 2 Tbsp butter
  \item 2 Tbsp oil
  \item Salt and pepper, to taste
  \item Fresh coriander leaves, to garnish
  \item Cooked basmati rice or naan bread, to serve
\end{ingredients}

\begin{method}
  \item Heat oil and butter in a large pan over medium heat. Add onion and cook until softened and golden.  
  \item Add garlic, ginger, and all spices (garam masala, cumin, paprika, turmeric, ground coriander). Stir for 1–2 minutes until fragrant.  
  \item Add chicken pieces and cook until lightly browned on all sides.  
  \item Stir in crushed tomatoes. Reduce heat and simmer for 15 minutes, until chicken is cooked through.  
  \item Stir in cream and simmer gently for another 5 minutes until the sauce is rich and creamy. Season with salt and pepper.  
  \item Serve hot, garnished with fresh coriander, alongside rice or naan bread.  
\end{method}

\begin{notes}
For extra flavour, marinate the chicken in yoghurt, lemon juice, and half of the spices for 30 minutes before cooking.  
This dish can also be made with coconut cream for a dairy-free option.  
\end{notes}
\end{recipebox}
