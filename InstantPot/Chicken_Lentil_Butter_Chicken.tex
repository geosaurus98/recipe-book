\begin{recipebox}{Butter Chicken-Style Chicken \& Lentil Curry}{4}{50 min}

\begin{ingredients}
  \item 1 tbsp oil
  \item 1 onion, diced
  \item 4 garlic cloves, minced
  \item 1 tbsp grated ginger
  \item 8 chicken drumsticks, skin removed
  \item 800g tinned diced tomatoes
  \item 150g red lentils, rinsed
  \item 2 tbsp garam masala
  \item 1 tsp ground turmeric
  \item 1 tbsp chicken stock powder
  \item 1 tsp salt
  \item \nicefrac{1}{2} tsp chilli flakes (optional)
  \item 400ml water
  \item 100ml light coconut cream (or coconut milk)
  \item 150g long grain brown rice (dry), to serve
\end{ingredients}

\begin{method}
  \item Set the Instant Pot to Saut\'e. Heat the oil and cook the onion for 3--4 min until softened. Add the garlic and ginger and cook for 1 min more.
  \item Add the drumsticks, tomatoes, lentils, garam masala, turmeric, stock powder, salt, chilli flakes and water. Stir to combine and make sure the lentils are submerged.
  \item Seal the lid with the valve on sealing. Pressure cook on high for 12 min, then natural release for 10 min before venting any remaining pressure.
  \item Remove the drumsticks, shred the meat off the bone, discard the bones and stir the meat back through the sauce.
  \item Set back to Saut\'e and stir in the coconut cream. Simmer uncovered for 5--10 min to thicken slightly. Taste and adjust the seasoning.
  \item Serve over a small portion (\nicefrac{1}{2}--\nicefrac{3}{4} cup cooked) of brown rice.
\end{method}

\begin{variationbox}{Stovetop Version}
  Saut\'e the onion, garlic and ginger in a large pot. Add everything except the coconut cream, cover and simmer on low for 40--45 min, stirring occasionally, until the chicken is cooked through and the lentils have broken down. Shred the chicken, stir in the coconut cream and simmer uncovered for 5--10 min.
\end{variationbox}

\begin{notelist}
  \item Water is reduced to 400ml (vs.\ 500ml for the slow cooker version) as pressure cooking loses no liquid to evaporation.
  \item The coconut cream goes in after pressure cooking, as thick, fatty ingredients can trigger a burn warning.
  \item Brown rice: pressure cook 1:1.25 rice to water on high for 20--22 min with a 10 min natural release, or simmer for 30--35 min.
  \item Freezes well. Portion into containers for meal prep.
\end{notelist}

\end{recipebox}
