\begin{recipebox}{Sausage \& Spinach Pasta}{6}{35 min}
  \begin{ingredients}
    \item 2 tsp olive oil
    \item 450\,g Italian sausages (casings removed)
    \item 1 medium onion, diced
    \item 3 garlic cloves, minced
    \item 1 jar (680\,g) marinara or pasta sauce
    \item 3 cups chicken stock
    \item 340\,g dry penne (or similar pasta)
    \item \nicefrac{1}{2} tsp salt (adjust if stock is salted)
    \item \nicefrac{1}{4} tsp black pepper
    \item 4 cups fresh baby spinach
    \item \nicefrac{1}{2} cup grated Parmesan cheese (for serving)
  \end{ingredients}

  \begin{method}
    \item Set Instant Pot to sauté mode. Add olive oil and sausage; cook until browned, breaking up with a spoon.
    \item Add onion and garlic; cook a few minutes until softened.
    \item Stir in marinara sauce, chicken stock, pasta, salt, and pepper. Mix well to submerge pasta.
    \item Cancel sauté. Secure lid, set valve to sealing. Pressure cook on HIGH for 5 minutes.
    \item Quick release steam. Stir in fresh spinach until wilted.
    \item Serve immediately, topped with Parmesan cheese.
  \end{method}
\end{recipebox}
