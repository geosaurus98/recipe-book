\begin{recipebox}{Chicken \& Rice}{6}{40 min}
  \begin{ingredients}
    \item 2 tbsp olive oil
    \item 1 medium onion, finely chopped
    \item 2 medium carrots, diced
    \item 2 celery ribs, diced
    \item 2 garlic cloves, minced
    \item 900\,g boneless, skinless chicken thighs (or breasts), cut into chunks
    \item 1\nicefrac{1}{2} tsp salt (or to taste)
    \item \nicefrac{1}{2} tsp black pepper
    \item 1 tsp paprika
    \item 2 cups long grain white rice, rinsed
    \item 4 cups chicken stock
    \item 2 tbsp fresh parsley, chopped (for garnish)
  \end{ingredients}

  \begin{method}
    \item Turn Instant Pot to sauté mode. Add oil, onion, carrot, celery, and garlic; cook until softened.
    \item Add chicken pieces, season with salt, pepper, and paprika. Sauté 5 minutes until lightly browned.
    \item Add rinsed rice and chicken stock. Stir to combine.
    \item Cancel sauté. Close lid, set valve to sealing. Pressure cook on HIGH for 10 minutes.
    \item Let pressure release naturally for 10 minutes, then quick release any remaining steam.
    \item Fluff rice gently, garnish with fresh parsley, and serve warm.
  \end{method}
\end{recipebox}
