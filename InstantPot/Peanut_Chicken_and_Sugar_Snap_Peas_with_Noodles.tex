\begin{recipebox}{Peanut Chicken \& Sugar Snap Peas with Noodles}{6}{More than 2 hrs}
  \begin{ingredients}
    \item 1\nicefrac{1}{4} lb chicken breast tenders, cubed
    \item 3 Tbsp cornstarch
    \item 2 Tbsp teriyaki sauce
    \item 2 tsp fresh garlic, minced
    \item \nicefrac{1}{4} tsp crushed red pepper
    \item 1 tsp dark sesame oil
    \item 2 cups chicken broth
    \item \nicefrac{1}{4} cup peanut butter
    \item 2\nicefrac{1}{2} cups sugar snap peas, trimmed
    \item 1 cup carrots, julienned
    \item 12 oz spaghetti
    \item \nicefrac{1}{2} cup green onion, sliced
    \item \nicefrac{1}{4} cup unsalted roasted peanuts, chopped
    \item Lime wedges, for serving
  \end{ingredients}

  \begin{method}
    \item In a bowl, toss the chicken with 2 Tbsp cornstarch, 1 Tbsp teriyaki sauce, 1 tsp garlic, and crushed red pepper.
    \item Heat a nonstick skillet over medium-high. Add oil, then the chicken mixture; brown for about 6 min. Stir in \nicefrac{1}{2} cup broth, scraping up browned bits. Transfer to Instant Pot.
    \item In another bowl, mix the remaining 1\nicefrac{1}{2} cups broth, peanut butter, 1 Tbsp cornstarch, remaining 1 Tbsp teriyaki sauce, and remaining 1 tsp garlic. Pour over the chicken.
    \item Close and lock lid. Select \textbf{Slow Cook}, “More” mode, set time to 1 hour 30 minutes.
    \item When done, stir in peas and carrots. Repeat slow cook for 30 minutes until veggies are crisp-tender.
    \item Meanwhile, cook spaghetti per package instructions (no salt/fat); drain.
    \item Add spaghetti to the Instant Pot and toss. Garnish with scallions, peanuts, and serve with lime wedges.
  \end{method}
\end{recipebox}
