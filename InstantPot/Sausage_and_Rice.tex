\begin{recipebox}{Sausage \& Rice}{6}{35 min}

% --- Base Ingredients ---
\begin{ingredients}
  \item 2 tbsp olive oil
  \item 450\,g pork or Italian sausages, crumbled
  \item 1 medium onion, diced
  \item 1 capsicum, diced
  \item 2 garlic cloves, minced
  \item 1 tsp Italian seasoning
  \item \nicefrac{1}{2} tsp salt
  \item \nicefrac{1}{4} tsp black pepper
  \item 1\nicefrac{1}{2} cups long-grain white rice
  \item 1 can (400\,g) diced tomatoes, with juices
  \item 3 cups chicken stock
  \item 1 cup shredded cheddar (optional)
\end{ingredients}

% --- Method ---
\begin{method}
  \item Set Instant Pot to \texttt{Sauté}. Heat oil, brown sausage.
  \item Add onion, pepper, garlic; cook until softened.
  \item Stir in seasoning, salt, pepper.
  \item Add rice, tomatoes, stock. Stir well.
  \item Cancel sauté. Seal lid. Cook \textbf{High Pressure, 7 min}.
  \item Natural release for 10 min, then quick release.
  \item Fluff and serve; add cheddar if using.
\end{method}

% --- Variations ---
\section*{Variations}

\begin{variationbox}{Cajun}
  \textbf{Seasoning swap:}
  \begin{itemize}
    \item 1\nicefrac{1}{2} tsp Cajun seasoning
    \item \nicefrac{1}{2} tsp smoked paprika
    \item \nicefrac{1}{4} tsp cayenne
  \end{itemize}
  \textbf{Optional:} 1 tsp hot sauce after cooking.
\end{variationbox}

\begin{variationbox}{Mexican}
  \textbf{Seasoning swap:}
  \begin{itemize}
    \item 1\nicefrac{1}{2} tsp cumin
    \item 1 tsp smoked paprika
    \item 1 tsp chili powder
    \item \nicefrac{1}{2} tsp oregano
  \end{itemize}
  \textbf{Optional additions:}
  \begin{itemize}
    \item \nicefrac{1}{2} cup corn (cook with tomatoes)
    \item Lime + coriander to serve
  \end{itemize}
\end{variationbox}

\begin{variationbox}{Mild Curry}
  \textbf{Seasoning swap:}
  \begin{itemize}
    \item 1 tbsp mild curry powder
    \item \nicefrac{1}{2} tsp turmeric
    \item \nicefrac{1}{2} cup coconut milk
  \end{itemize}
  \textbf{Optional:} Add coconut milk with stock.
\end{variationbox}

\end{recipebox}
