\begin{recipebox}{Berry \& Oat Power Shake}{1}{--}
\begin{ingredients}
  \item 1 scoop vanilla protein powder
  \item \textonehalf{} cup frozen mixed berries
  \item \textonehalf{} cup rolled oats
  \item 200g Greek yogurt
  \item 250ml milk
  \item 1 tsp honey
  \item 4--5 ice cubes
\end{ingredients}

\begin{method}
  \item Add all ingredients to a blender.
  \item Blend on high for 45--60 seconds until completely smooth (oats need a bit longer).
  \item Pour and drink immediately --- this one thickens quickly.
\end{method}

\begin{notelist}
  \item Approx. \$1.50/shake, 400 kcal, 36g protein per serve.
  \item \textbf{Tip:} The oats make this shake very filling --- almost meal-replacement level. Great as a post-walk breakfast if you're short on time.
\end{notelist}
\end{recipebox}
