\begin{recipebox}{Classic Chocolate Peanut Butter Shake}{1}{--}
\begin{ingredients}
  \item 1 scoop chocolate or vanilla protein powder
  \item 1 tbsp natural peanut butter
  \item 1 banana (fresh or frozen for extra thickness)
  \item 300\,ml milk
  \item 1 tsp cocoa powder
  \item 4--5 ice cubes
\end{ingredients}

\begin{method}
  \item Add all ingredients to a blender.
  \item Blend on high for 30--45 seconds until smooth and creamy.
  \item Pour into a large glass and drink immediately.
\end{method}

\begin{notelist}
  \item Approx. \$1.80/shake, 380 kcal, 38\,g protein per serve.
  \item \textbf{Tip:} Frozen bananas make the shake thicker and creamier. Peel and freeze overripe bananas in a zip-lock bag --- they last months and are perfect for shakes.
\end{notelist}
\end{recipebox}
