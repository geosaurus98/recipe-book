\begin{recipebox}{Green Spinach \& Banana Shake}{1}{--}
\begin{ingredients}
  \item 1 scoop vanilla protein powder
  \item 1 large handful of baby spinach
  \item 1 banana (frozen for best texture)
  \item 250ml milk or almond milk
  \item 1 tbsp chia seeds
  \item 1 tsp honey
  \item 4--5 ice cubes
\end{ingredients}

\begin{method}
  \item Add spinach and milk to the blender first. Blend briefly to break down the spinach.
  \item Add remaining ingredients and blend until smooth.
  \item Pour and drink immediately.
\end{method}

%  HEALTHY DESSERTS \& SWEET TREATS

\begin{notelist}
  \item Approx. \$1.40/shake, 330 kcal, 30g protein per serve.
  \item \textbf{Tip:} Don't let the green colour put you off --- you can't taste the spinach at all. It adds a significant micronutrient boost with minimal calories. A great way to sneak in extra vegetables.
\end{notelist}
\end{recipebox}
