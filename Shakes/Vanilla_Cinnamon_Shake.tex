\begin{recipebox}{Vanilla Cinnamon Shake}{1}{--}
\begin{ingredients}
  \item 1 scoop vanilla protein powder
  \item 300ml milk
  \item 1 tsp cinnamon
  \item \textonehalf{} tsp vanilla essence
  \item 1 tsp honey or maple syrup
  \item 4--5 ice cubes
\end{ingredients}

\begin{method}
  \item Add all ingredients to a blender or shaker bottle.
  \item Blend or shake vigorously until combined.
  \item Pour over ice and drink immediately.
\end{method}

\begin{notelist}
  \item Approx. \$1.20/shake, 310 kcal, 32g protein per serve.
  \item \textbf{Tip:} The simplest shake in the book --- works in a shaker bottle with no blender needed. Great for days when you just want something quick and simple.
\end{notelist}
\end{recipebox}
