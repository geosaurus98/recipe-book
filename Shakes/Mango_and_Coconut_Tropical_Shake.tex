\begin{recipebox}{Mango \& Coconut Tropical Shake}{1}{--}
\begin{ingredients}
  \item 1 scoop vanilla protein powder
  \item 1 cup frozen mango chunks
  \item 200\,ml coconut milk (light)
  \item 100\,ml water or milk
  \item Juice of \textonehalf{} lime
  \item 4--5 ice cubes
\end{ingredients}

\begin{method}
  \item Add all ingredients to a blender.
  \item Blend until completely smooth.
  \item Pour into a glass and serve immediately.
\end{method}

\begin{notelist}
  \item Approx. \$1.60/shake, 350 kcal, 32\,g protein per serve.
  \item \textbf{Tip:} Frozen mango is much cheaper than fresh and works better in shakes. Available in large bags at Pak'nSave. Swap mango for frozen pineapple for variety.
\end{notelist}
\end{recipebox}
