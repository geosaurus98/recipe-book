\begin{recipebox}{Baked Cinnamon Apples}{2}{10 min prep + 30 min}
\begin{ingredients}
  \item 2 large apples, cored
  \item 2 tbsp rolled oats
  \item 1 tbsp honey
  \item 1 tsp cinnamon
  \item 1 tsp butter
  \item Pinch of nutmeg
  \item \textbf{To serve:} Greek yoghurt
\end{ingredients}

\begin{method}
  \item Preheat oven to 180\textdegree C.
  \item Core apples, leaving the base intact. Place in a small baking dish.
  \item Mix oats, honey, cinnamon, butter, and nutmeg together. Spoon into the hollow of each apple.
  \item Add a splash of water to the base of the dish.
  \item Bake for 25--30 minutes until apples are tender and filling is golden.
  \item Serve warm with a generous dollop of Greek yoghurt.
\end{method}

\begin{notelist}
  \item \textbf{Tip:} Feels indulgent but is incredibly wholesome. Royal Gala or Granny Smith apples work best. A great dessert for cold evenings.
\end{notelist}
\end{recipebox}
