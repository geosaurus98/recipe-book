\begin{recipebox}{Sticky Date Pudding}{8}{1 hr}
\begin{ingredients}
  \item[] \textbf{Pudding}
  \item 200g pitted dates, chopped
  \item 1 cup boiling water
  \item 1 tsp baking soda
  \item 60g butter, softened
  \item \nicefrac{3}{4} cup brown sugar
  \item 2 eggs
  \item 1 tsp vanilla essence
  \item 1\nicefrac{1}{4} cups self-raising flour
  \item[] \textbf{Butterscotch sauce}
  \item 100g butter
  \item 1 cup brown sugar
  \item 300ml cream
\end{ingredients}

\begin{method}
  \item Preheat oven to 180\degree C bake. Grease and line a 20cm square or round cake tin.
  \item Put the dates in a bowl, pour over the boiling water and stir in the baking soda (it will foam). Leave for 10 minutes to soften.
  \item Cream the butter and brown sugar, then beat in the eggs one at a time and the vanilla.
  \item Fold in the flour, then the dates and all their liquid. The batter will be quite runny.
  \item Pour into the tin and bake for 35--40 minutes, until a skewer comes out clean.
  \item Meanwhile, make the sauce: melt the butter and brown sugar in a saucepan, stir in the cream and simmer for 3--4 minutes until slightly thickened.
  \item Poke holes in the warm pudding with a skewer and pour over a little sauce. Serve in squares with the rest of the sauce.
\end{method}

\begin{notes}
The pudding freezes well in portions. Reheat a square in the microwave and pour over warm sauce.
\end{notes}
\end{recipebox}
