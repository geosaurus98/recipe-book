\begin{recipebox}{Frozen Yoghurt Bark}{8}{10 min (plus 3 hours freezing)}
\begin{ingredients}
  \item 500\,g Greek yoghurt
  \item 2 tbsp honey
  \item 1 tsp vanilla essence
  \item \textonehalf{} cup mixed berries
  \item 2 tbsp dark chocolate chips
  \item 2 tbsp granola or crushed almonds
\end{ingredients}

\begin{method}
  \item Line a baking tray with baking paper.
  \item Mix Greek yoghurt, honey, and vanilla together. Spread evenly over the tray about 1cm thick.
  \item Scatter berries, chocolate chips, and granola over the top.
  \item Freeze for at least 3 hours until completely solid.
  \item Break into rough pieces and store in a zip-lock bag in the freezer.
\end{method}

\begin{notelist}
  \item \textbf{Tip:} Keeps in the freezer for up to 2 months. A brilliant alternative to ice cream on warm Christchurch evenings.
\end{notelist}
\end{recipebox}
