\begin{recipebox}{Beef Schnitzel Roll}{4}{45 min}
  \begin{ingredients}
    \item 4 beef schnitzels (or chicken schnitzels)
    \item 4 slices Edam cheese
    \item 6 cloves garlic
    \item Fresh rosemary
    \item 2--3 slices streaky bacon
    \item 1\,cup flour (for dusting)
    \item \nicefrac{1}{2} glass white wine
    \item Olive oil
    \item Salt and pepper, to taste
    \item Toothpicks
    \item Non-stick frypan with lid
  \end{ingredients}

  \begin{method}
    \item \textbf{Prepare the pesto:} Crush the garlic and rosemary together in a mortar and pestle to form a coarse paste.
    \item \textbf{Marinate the meat:} Cut each schnitzel in half. Brush half of the slices with a little of the garlic–rosemary pesto and let marinate for 30 minutes. Afterward, gently remove excess pesto.
    \item \textbf{Assemble the rolls:} On each marinated schnitzel half, place a slice of cheese and half a slice of bacon. Cover with a second schnitzel half and roll tightly, securing with a toothpick.  
    Alternatively, layer the two pieces flat (like a sandwich) and secure with toothpicks.
    \item \textbf{Coat the rolls:} Lightly dust each roll in flour.
    \item \textbf{Sear the rolls:} In a non-stick frypan, heat olive oil with a little garlic and rosemary until fragrant and golden. Add the schnitzel rolls, season with salt and pepper, and brown for about 2 minutes per side.
    \item \textbf{Deglaze and simmer:} Pour in the white wine and simmer uncovered for 4 minutes. Cover the pan with a lid and cook gently for another 5 minutes.
    \item \textbf{To serve:} Serve hot with crusty bread and seasonal vegetables.
  \end{method}

  \begin{notes}
    This dish works equally well with chicken schnitzel for a lighter version.  
    For extra flavour, add a dash of lemon juice before serving or sprinkle with chopped parsley.  
    The melted cheese and bacon make the rolls rich and savoury — perfect for a hearty dinner.
  \end{notes}
\end{recipebox}
