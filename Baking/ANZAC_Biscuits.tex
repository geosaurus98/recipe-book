\begin{recipebox}{ANZAC Biscuits (Chewy)}{\textasciitilde20 biscuits}{30 min}
\begin{ingredients}
  \item 1 cup rolled oats
  \item 1 cup plain flour
  \item 1 cup brown sugar (packed)
  \item \nicefrac{3}{4} cup desiccated coconut
  \item 125g butter
  \item 2 Tbsp golden syrup
  \item 1 tsp baking soda
  \item 2 Tbsp boiling water
\end{ingredients}

\begin{method}
  \item Preheat oven to 180\degree C bake. Line two trays with baking paper.
  \item Mix the oats, flour, brown sugar and coconut in a large bowl.
  \item Melt the butter and golden syrup together in a saucepan (or the microwave).
  \item Dissolve the baking soda in the boiling water and stir it into the butter mixture --- it will foam up.
  \item Pour the foaming mixture into the dry ingredients and mix well.
  \item Roll tablespoons of mixture into balls, place on the trays about 5cm apart and flatten slightly with a fork.
  \item Bake for 10--12 minutes, until golden at the edges but still soft in the middle.
  \item Leave on the tray for 5 minutes to firm up, then cool on a wire rack.
\end{method}

\begin{notes}
Brown sugar and a short bake keep them chewy. For crunchy ANZACs, use white sugar and bake for 15 minutes.
Keeps for a week in an airtight tin.
\end{notes}
\end{recipebox}
