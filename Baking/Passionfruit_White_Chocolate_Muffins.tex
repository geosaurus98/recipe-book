\begin{recipebox}{Passionfruit \& White Chocolate Muffins}{12}{35 min}
\begin{ingredients}
  \item 2 cups plain flour
  \item 4 tsp baking powder
  \item \nicefrac{1}{4} tsp salt
  \item \nicefrac{3}{4} cup white sugar
  \item \nicefrac{3}{4} cup white chocolate chips
  \item 1 egg
  \item \nicefrac{3}{4} cup milk
  \item \nicefrac{1}{4} cup passionfruit pulp (about 4 passionfruit, or tinned pulp)
  \item 100g butter, melted
  \item 1 tsp vanilla essence
\end{ingredients}

\subsection*{Passionfruit Drizzle}
\begin{itemize}[leftmargin=2em]
  \item \nicefrac{3}{4} cup icing sugar
  \item 1--2 Tbsp passionfruit pulp
\end{itemize}

\begin{method}
  \item Preheat oven to 200\degree C bake. Grease or line a 12-hole muffin tin.
  \item Sift the flour, baking powder, salt and sugar into a bowl. Stir in the white chocolate chips.
  \item Whisk the egg, milk, passionfruit pulp, melted butter and vanilla together.
  \item Pour into the dry ingredients and fold until just combined.
  \item Spoon into the tin and bake for 15--18 minutes, until golden and a skewer comes out clean.
  \item Cool on a wire rack. Mix the icing sugar with enough passionfruit pulp to make a thick drizzle and spoon over the muffins.
\end{method}

\begin{notes}
Tinned or tubbed passionfruit pulp is cheaper than fresh most of the year and works just as well.
\end{notes}
\end{recipebox}
