\begin{recipebox}{Yoghurt Naan Bread}{6}{30 min}
\begin{ingredients}
  \item 2 cups flour
  \item \nicefrac{1}{2} tsp baking powder
  \item 1 tsp salt
  \item 1 cup natural unsweetened yoghurt
  \item 2 tbsp oil (plus extra for frying)
\end{ingredients}

\begin{method}
  \item In a large bowl, mix together the flour, baking powder, and salt.  
  \item Add the yoghurt and oil. Stir until a dough forms.  
  \item Turn dough onto a floured surface and knead for 2–3 minutes until smooth.  
  \item Divide dough into 6–8 equal pieces. Roll each piece into a flat oval or circle about \nicefrac{1}{2} cm thick.  
  \item Heat a drizzle of oil in a frying pan over medium-high heat.  
  \item Cook naan for 1–2 minutes per side until golden brown spots appear and bread puffs slightly.  
  \item Keep warm in a clean tea towel while cooking the rest. Serve hot with curries, dips, or spreads.  
\end{method}

\begin{notes}
Brush naan with melted butter and garlic for a garlic naan variation.  
This recipe is yeast-free and quick to prepare.  
Pairs perfectly with curries such as Butter Chicken.  
\end{notes}
\end{recipebox}
