\begin{recipebox}{Cheese Scones}{10--12}{25 min}
\begin{ingredients}
  \item 3 cups plain flour
  \item 6 tsp baking powder
  \item \nicefrac{1}{4} tsp salt
  \item Pinch of cayenne pepper
  \item 1\nicefrac{1}{2} cups grated tasty cheddar
  \item 1--1\nicefrac{1}{2} cups milk
\end{ingredients}

\begin{method}
  \item Preheat oven to 220\degree C bake. Lightly flour a baking tray.
  \item Sift the flour, baking powder, salt and cayenne into a bowl. Stir in the cheese.
  \item Add 1 cup of milk and mix quickly with a knife to a soft, slightly sticky dough, adding more milk if needed.
  \item Tip onto a floured bench and pat (don't knead) into a rectangle about 2cm thick.
  \item Cut into 10--12 squares and place on the tray just touching.
  \item Bake for 10--12 minutes, until golden and they sound hollow when tapped underneath.
  \item Wrap in a clean tea towel to keep soft. Best eaten warm with butter.
\end{method}

\begin{notes}
The less you handle the dough, the lighter the scones. Mix just until it comes together.
Sprinkle a little extra cheese on top before baking for a crunchy crust.
\end{notes}
\end{recipebox}
