\begin{recipebox}{Ham \& Cheese Breakfast Muffins}{12}{30 min}
\begin{ingredients}
  \item 2 cups self-raising flour
  \item 1 cup grated tasty cheddar
  \item 150\,g ham, diced
  \item 2 spring onions, sliced
  \item \nicefrac{1}{2} tsp mustard powder (optional)
  \item Black pepper
  \item 2 eggs
  \item 1 cup milk
  \item \nicefrac{1}{4} cup neutral oil
\end{ingredients}

\begin{method}
  \item Preheat oven to 200\degree C bake. Grease or line a 12-hole muffin tin.
  \item Mix the flour, cheese, ham, spring onion, mustard powder and pepper in a bowl.
  \item Whisk the eggs, milk and oil together, pour into the dry ingredients and fold until just combined.
  \item Spoon into the tin and bake for 15--18 minutes, until golden and a skewer comes out clean.
\end{method}

\begin{notes}
Make a batch on Sunday for grab-and-go breakfasts. Freeze, then microwave for 30 seconds from frozen.
A handful of chopped spinach or diced capsicum can go in with the ham.
\end{notes}
\end{recipebox}
