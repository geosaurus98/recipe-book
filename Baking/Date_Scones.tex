\begin{recipebox}{Date Scones}{10--12}{25 min}
\begin{ingredients}
  \item 3 cups plain flour
  \item 6 tsp baking powder
  \item \nicefrac{1}{4} tsp salt
  \item 2 Tbsp white sugar
  \item 75g cold butter, cubed
  \item 1 cup pitted dates, chopped
  \item 1--1\nicefrac{1}{2} cups milk
\end{ingredients}

\begin{method}
  \item Preheat oven to 220\degree C bake. Lightly flour a baking tray.
  \item Sift the flour, baking powder and salt into a bowl and stir in the sugar.
  \item Rub in the butter with your fingertips until it looks like fine breadcrumbs. Stir in the dates.
  \item Add 1 cup of milk and mix quickly with a knife to a soft dough, adding more milk if needed.
  \item Tip onto a floured bench and pat (don't knead) into a rectangle about 2cm thick.
  \item Cut into 10--12 squares and place on the tray just touching.
  \item Bake for 10--12 minutes, until golden and they sound hollow when tapped underneath.
\end{method}

\begin{notes}
Toss the chopped dates in a spoonful of the flour first so they don't clump together.
\end{notes}
\end{recipebox}
