\begin{recipebox}{Date Scones}{12}{25 min}
\begin{ingredients}
  \item 3 cups plain flour
  \item 6 tsp baking powder
  \item Pinch of salt
  \item 1 Tbsp white sugar
  \item \nicefrac{1}{2} tsp ground cinnamon
  \item 75g cold butter, cubed
  \item \nicefrac{3}{4} cup pitted dates, chopped
  \item 1--1\nicefrac{1}{2} cups milk
  \item Cinnamon sugar, to sprinkle (optional)
\end{ingredients}

\begin{method}
  \item Preheat oven to 220\degree C bake. Grease or flour a baking tray.
  \item Sift the flour, baking powder and salt into a bowl and stir in the sugar and cinnamon.
  \item Rub in the butter with your fingertips until it looks like fine breadcrumbs. Stir in the dates.
  \item Add 1 cup of milk and mix quickly with a knife to a soft, slightly sticky dough, adding more milk if needed.
  \item Tip onto the floured tray and pat (don't knead) to about 2cm thick. Cut into 12 pieces and separate them slightly.
  \item Sprinkle with cinnamon sugar, if using, and bake for about 10 minutes, until golden.
\end{method}

\begin{notes}
Quantities follow the Edmonds date scone recipe.
Toss the chopped dates in a spoonful of the flour first so they don't clump together.
\end{notes}
\end{recipebox}
