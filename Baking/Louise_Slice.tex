\begin{recipebox}{Louise Slice}{\textasciitilde24 pieces}{50 min}
\begin{ingredients}
  \item[] \textbf{Base}
  \item 125g butter, softened
  \item \nicefrac{1}{4} cup white sugar
  \item 2 egg yolks (keep the whites for the topping)
  \item 1\nicefrac{1}{2} cups plain flour
  \item 1 tsp baking powder
  \item \nicefrac{1}{2} cup raspberry jam
  \item[] \textbf{Coconut meringue topping}
  \item 2 egg whites
  \item \nicefrac{1}{2} cup white sugar
  \item 1 cup desiccated coconut
\end{ingredients}

\begin{method}
  \item Preheat oven to 180\degree C bake. Grease and line a 20 $\times$ 30cm slice tin.
  \item Cream the butter and sugar, then beat in the egg yolks.
  \item Sift in the flour and baking powder and mix to a firm dough. Press evenly into the tin.
  \item Spread the jam over the base.
  \item Beat the egg whites until stiff, then gradually beat in the sugar until thick and glossy. Fold in the coconut.
  \item Spread the topping over the jam, right to the edges.
  \item Bake for 25--30 minutes, until the topping is lightly golden.
  \item Cool in the tin before cutting into squares.
\end{method}

\begin{notes}
Use a clean, dry bowl for the egg whites --- any trace of yolk or grease stops them whipping.
\end{notes}
\end{recipebox}
