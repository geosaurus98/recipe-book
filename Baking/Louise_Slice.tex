\begin{recipebox}{Louise Slice}{\textasciitilde24 pieces}{50 min}
\begin{ingredients}
  \item[] \textbf{Base}
  \item 125\,g butter, softened
  \item \nicefrac{3}{4} cup caster sugar
  \item 2 egg yolks (keep the whites for the topping)
  \item 1\nicefrac{3}{4} cups plain flour
  \item 1\nicefrac{1}{2} tsp baking powder
  \item \nicefrac{1}{2} cup raspberry jam
  \item[] \textbf{Coconut meringue topping}
  \item 2 egg whites
  \item \nicefrac{1}{4} cup caster sugar
  \item 1 cup long-thread or desiccated coconut, plus extra to sprinkle
  \item 1 tsp vanilla essence
\end{ingredients}

\begin{method}
  \item Preheat oven to 170\degree C bake. Line a 20 $\times$ 30cm slice tin with baking paper.
  \item Cream the butter and sugar, then beat in the egg yolks.
  \item Sift in the flour and baking powder and mix --- it will be crumbly. Press firmly and evenly into the tin.
  \item Bake for 10 minutes. Raise the oven to 180\degree C.
  \item Meanwhile, beat the egg whites to stiff peaks, then gradually beat in the sugar until thick and glossy. Fold in the coconut and vanilla.
  \item Spread the jam over the base, then spread the meringue over the jam right to the edges. Sprinkle with a little extra coconut.
  \item Bake for 10--15 minutes, until the topping is lightly golden.
  \item Cool in the tin before cutting into squares.
\end{method}

\begin{notes}
Quantities follow Chelsea Sugar's Louise Cake recipe.
Use a clean, dry bowl for the egg whites --- any trace of yolk or grease stops them whipping.
\end{notes}
\end{recipebox}
