\begin{recipebox}{Carrot Cake Muffins with Cream Cheese Drizzle}{12}{40 min}
\begin{ingredients}
  \item 2 cups plain flour
  \item 2 tsp baking powder
  \item 1 tsp baking soda
  \item 2 tsp ground cinnamon
  \item 1 tsp mixed spice
  \item \nicefrac{1}{4} tsp salt
  \item 1 cup brown sugar
  \item 2 eggs
  \item \nicefrac{1}{2} cup neutral oil
  \item \nicefrac{1}{2} cup milk or plain yoghurt
  \item 1 tsp vanilla essence
  \item 2 cups grated carrot (about 2 medium)
  \item \nicefrac{1}{2} cup sultanas
  \item \nicefrac{1}{2} cup chopped walnuts (optional)
\end{ingredients}

\subsection*{Cream Cheese Drizzle}
\begin{itemize}[leftmargin=2em]
  \item 100g cream cheese, softened
  \item \nicefrac{1}{2} cup icing sugar
  \item 1--2 Tbsp milk
  \item \nicefrac{1}{2} tsp vanilla essence
\end{itemize}

\begin{method}
  \item Preheat oven to 180\degree C bake. Grease or line a 12-hole muffin tin.
  \item Sift the flour, baking powder, baking soda, spices and salt into a bowl. Stir in the brown sugar.
  \item Whisk the eggs, oil, milk and vanilla together.
  \item Add the wet ingredients, carrot, sultanas and walnuts to the dry and fold until just combined.
  \item Spoon into the tin and bake for 20--22 minutes, until a skewer comes out clean.
  \item Cool completely.
  \item Beat the cream cheese, icing sugar and vanilla until smooth, adding milk until it's runny enough to drizzle. Drizzle over the muffins.
\end{method}

\begin{notes}
Store iced muffins in the fridge because of the cream cheese. Un-iced muffins freeze well --- drizzle after thawing.
\end{notes}
\end{recipebox}
