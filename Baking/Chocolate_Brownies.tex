\begin{recipebox}{Chocolate Brownies}{16 pieces}{45 min}
\begin{ingredients}
  \item 175g butter
  \item 200g dark chocolate, chopped
  \item 1\nicefrac{1}{4} cups white sugar
  \item 3 eggs
  \item 1 tsp vanilla essence
  \item \nicefrac{3}{4} cup plain flour
  \item \nicefrac{1}{4} cup cocoa powder
  \item Pinch of salt
  \item \nicefrac{1}{2} cup chocolate chips (optional)
\end{ingredients}

\begin{method}
  \item Preheat oven to 180\degree C bake. Grease and line a 20cm square tin.
  \item Melt the butter and chocolate together in a saucepan over low heat (or in the microwave in short bursts), stirring until smooth. Cool for 5 minutes.
  \item Whisk in the sugar, then the eggs one at a time, then the vanilla.
  \item Sift in the flour, cocoa and salt and fold until just combined. Fold in the chocolate chips, if using.
  \item Pour into the tin and bake for 25--30 minutes, until the top is set but a skewer comes out with moist crumbs.
  \item Cool completely in the tin before cutting.
\end{method}

\begin{notes}
Take them out while still slightly underdone in the middle --- they firm up as they cool. Overbaked brownies turn cakey.
\end{notes}
\end{recipebox}
