\begin{recipebox}{Afghan Biscuits}{\textasciitilde24 biscuits}{45 min}
\begin{ingredients}
  \item 200\,g butter, softened
  \item \nicefrac{1}{2} cup white sugar
  \item 1\nicefrac{1}{4} cups plain flour
  \item \nicefrac{1}{4} cup cocoa powder
  \item 2 cups cornflakes
\end{ingredients}

\subsection*{Chocolate Icing}
\begin{itemize}[leftmargin=2em]
  \item 1 cup icing sugar
  \item 2 tbsp cocoa powder
  \item 1 tsp butter, softened
  \item 2--3 tbsp boiling water
  \item A few drops of vanilla essence
  \item Walnut halves, to decorate
\end{itemize}

\begin{method}
  \item Preheat oven to 180\degree C bake. Line two trays with baking paper.
  \item Cream the butter and sugar until light and fluffy.
  \item Sift in the flour and cocoa and mix well, then gently fold in the cornflakes --- don't crush them too much.
  \item Roll tablespoons of mixture into balls, place on the trays and flatten slightly.
  \item Bake for 15 minutes. Cool on a wire rack.
  \item For the icing, sift the icing sugar and cocoa into a bowl, add the butter, vanilla and enough boiling water to make a thick, spreadable icing.
  \item Ice the cooled biscuits and press a walnut half on top of each.
\end{method}

\begin{notes}
Quantities follow the Edmonds recipe. The biscuits must be completely cold before icing, or the icing will slide off.
Use a name-brand cornflake if you can --- Edmonds bakers report budget cornflakes turn hard and chewy.
\end{notes}
\end{recipebox}
