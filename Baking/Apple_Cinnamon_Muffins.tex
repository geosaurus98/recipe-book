\begin{recipebox}{Apple Cinnamon Muffins}{12}{35 min}
\begin{ingredients}
  \item 2 cups plain flour
  \item 4 tsp baking powder
  \item \nicefrac{1}{4} tsp salt
  \item \nicefrac{3}{4} cup white sugar
  \item 2 tsp ground cinnamon
  \item 2 apples, peeled, cored and diced (about 2 cups)
  \item 1 egg
  \item 1 cup milk
  \item 100g butter, melted
  \item 1 tsp vanilla essence
\end{ingredients}

\subsection*{Cinnamon Sugar Topping}
\begin{itemize}[leftmargin=2em]
  \item 2 Tbsp white sugar
  \item \nicefrac{1}{2} tsp ground cinnamon
\end{itemize}

\begin{method}
  \item Preheat oven to 200\degree C bake. Grease or line a 12-hole muffin tin.
  \item Sift the flour, baking powder, salt, sugar and cinnamon into a bowl. Stir in the diced apple so it's coated in flour.
  \item Whisk the egg, milk, melted butter and vanilla together in a jug.
  \item Pour the wet ingredients into the dry and fold until just combined --- a few lumps are fine.
  \item Spoon into the tin and sprinkle with the cinnamon sugar.
  \item Bake for 15--18 minutes, until risen and a skewer comes out clean.
  \item Cool in the tin for 5 minutes before moving to a wire rack.
\end{method}

\begin{notes}
Coating the apple in the flour mixture stops it sinking to the bottom.
Over-mixing makes muffins tough --- stop as soon as there's no dry flour.
\end{notes}
\end{recipebox}
