\begin{recipebox}{Apple \& Feijoa Crumble Muffins}{12}{40 min}
\begin{ingredients}
  \item 2 cups plain flour
  \item 4 tsp baking powder
  \item \nicefrac{1}{4} tsp salt
  \item \nicefrac{3}{4} cup white sugar
  \item 1 tsp ground cinnamon
  \item 1 apple, peeled, cored and diced
  \item 1 cup chopped feijoa flesh (6--8 feijoas)
  \item 1 egg
  \item 1 cup milk
  \item 100\,g butter, melted
\end{ingredients}

\subsection*{Crumble Topping}
\begin{itemize}[leftmargin=2em]
  \item 3 tbsp plain flour
  \item 3 tbsp brown sugar
  \item 3 tbsp rolled oats
  \item 30\,g cold butter, cubed
\end{itemize}

\begin{method}
  \item Preheat oven to 200\degree C bake. Grease or line a 12-hole muffin tin.
  \item Make the crumble: rub the butter into the flour, brown sugar and oats with your fingertips until it looks like rough crumbs. Set aside.
  \item Sift the flour, baking powder, salt, sugar and cinnamon into a bowl. Stir in the apple and feijoa.
  \item Whisk the egg, milk and melted butter together, pour into the dry ingredients and fold until just combined.
  \item Spoon into the tin and scatter the crumble over the top, pressing it on lightly.
  \item Bake for 18--20 minutes, until golden and a skewer comes out clean.
  \item Cool in the tin for 5 minutes before moving to a wire rack.
\end{method}

\begin{notes}
Out of feijoa season, use 2 apples, or 1 apple and 1 cup frozen berries (added from frozen).
\end{notes}
\end{recipebox}
