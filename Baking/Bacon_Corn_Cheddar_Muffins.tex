\begin{recipebox}{Bacon, Corn \& Cheddar Muffins}{12}{35 min}
\begin{ingredients}
  \item 3 rashers bacon, diced
  \item 2 cups self-raising flour
  \item 1 cup grated tasty cheddar (plus extra to top)
  \item 1 cup whole kernel corn, drained (canned or frozen)
  \item 2 spring onions, sliced
  \item Black pepper
  \item 1 egg
  \item 1 cup milk
  \item 50\,g butter, melted
\end{ingredients}

\begin{method}
  \item Preheat oven to 200\degree C bake. Grease or line a 12-hole muffin tin.
  \item Fry the bacon until crisp. Drain on paper towels and cool.
  \item Mix the flour, cheese, corn, spring onion, bacon and a good grind of pepper in a bowl.
  \item Whisk the egg, milk and melted butter together, pour into the dry ingredients and fold until just combined.
  \item Spoon into the tin and sprinkle with a little extra cheese.
  \item Bake for 15--18 minutes, until golden and a skewer comes out clean.
\end{method}

\begin{notes}
Great for lunchboxes. They freeze well --- reheat for 20--30 seconds in the microwave.
\end{notes}
\end{recipebox}
