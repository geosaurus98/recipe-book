
\begin{recipebox}{Homemade Burgers \& Fries}{2}{40 min}
\begin{ingredients}
  \item 500 g lean beef mince
  \item 1 onion, finely chopped
  \item 1 egg
  \item \nicefrac{1}{2} cup breadcrumbs (or oats)
  \item 1 tbsp Worcestershire Sauce
  \item 1 tbsp Ketchup
  \item Salt, pepper
  \item 4-6 burger buns
  \item Lettuce, tomato, cheese, gherkins, red onion
  \item 2 medium sweet potatoes, cut into wedges
  \item Olive oil spray
\end{ingredients}

\begin{method}
  \item Preheat oven to 200°C. Toss wedges with olive oil, bake 25–30 min.
  \item To a large bowl, add the beef mince, onion, egg, breadcrumbs, Worcestershire sauce, ketchup, salt and pepper. Use your hands to combine well.
  \item Divide the mixture into 4–6 portions, then use your hands to form patties. Their size ideally needs to be just bigger than the burger buns you are using as the patties will shrink slightly as they cook.
  \item Use your thumb to make an indent in one side of the patty. This helps the patties keep their shape during cooking.
  \item Heat the olive oil in a heavy-based frying pan over medium–high heat. Cook the burger patties for 10–12 minutes, turning once halfway through. 
  \item Add the water to the pan in the last 2 minutes of cooking. This step is optional, it allows you to scrape up all of the delicious sticky bits in the base of the pan and they cling to the outside of the patties = more flavour! 
  \item Serve the buns topped with burger patties, cheese, tomato, red onion, gherkins, lettuce, mayonnaise, ketchup and mustard. Serve with fries.
\end{method}
\textbf{Source:} \href{https://simplehomeedit.com/recipe/homemade-beef-burger/}{Juicy Homemade Beef Burgers}

\begin{notes}
For a delicious chargrilled result, you can cook the patties in a chargrill pan or cook them on a barbecue plate. Heat the plate to medium–high, spray the plate with oil just before cooking and cook the patties for 10–12 minutes, turning once halfway.
\end{notes}

\end{recipebox}

\begin{recipebox}{Old School Beef Tacos}{4}{30 min}
\begin{ingredients}
  \item 1 Tbsp oil
  \item 2 garlic cloves, minced
  \item 1 onion, finely chopped
  \item 500 g beef mince
  \item 2 Tbsp tomato paste
  \item \nicefrac{1}{4} cup (65 mL) water
\end{ingredients}

\subsection*{Homemade Taco Seasoning}
\begin{itemize}[leftmargin=2em]
  \item 1 tsp garlic powder
  \item 1 tsp onion powder
  \item 1 tsp dried oregano
  \item 2 tsp cumin powder
  \item 2 tsp paprika
  \item \nicefrac{1}{4} tsp cayenne pepper (optional, adjust to taste)
  \item 1 tsp salt
  \item \nicefrac{1}{4} tsp black pepper
\end{itemize}

\subsection*{Toppings}
\begin{itemize}[leftmargin=2em]
  \item 10–12 crispy taco shells or soft tortillas
  \item Shredded cheese (cheddar, tasty, or Monterey Jack if available)
  \item Shredded iceberg lettuce
  \item 2 tomatoes, chopped
  \item \nicefrac{1}{2} red or white onion, chopped
  \item Taco sauce or salsa (optional)
  \item Fresh coriander, jalapeños, or hot sauce (optional)
\end{itemize}

\begin{method}
  \item Preheat oven to 180\degree C.
  \item Heat oil in a large skillet over high heat. Add garlic and onion, cooking for 2 minutes until golden.
  \item Add beef and cook for 2 minutes, breaking it up until browned.
  \item Add taco seasonings and cook for 2 minutes until beef is cooked through.
  \item Stir in tomato paste and water. Cook for 1 minute until thickened and juicy (not watery).
  \item Place taco shells in a baking pan. Fill with taco meat and top with cheese. Bake 5–7 minutes until cheese is melted and shells are crispy. (Or, warm shells separately following packet instructions.)
  \item Serve hot with lettuce, tomato, onion, sour cream, and other toppings of choice.
\end{method}

\begin{notes}
For a quick Kiwi version, use a 35 g packet of taco seasoning and a 400 g tin of diced tomatoes (Woolworths NZ method).
\end{notes}

\textbf{Source:} \href{https://www.recipetineats.com/ground-beef-tacos-recipe/}{recipetineats – Beef Tacos}
\end{recipebox}

\begin{recipebox}{Chicken Thai Green Curry}{4 servings}{Prep 5 min, Cook 25 min (Total 30 min)}
\begin{ingredients}
  \item 1 tsp sesame oil
  \item 600 g diced chicken (breast or thigh)
  \item 1 tsp grated ginger
  \item 2 tbsp green curry paste
  \item 1 can (400 mL) coconut cream
  \item 1 can (227 g, drained) bamboo shoots
  \item 1 red capsicum, sliced
  \item $\tfrac{1}{2}$ cup chicken stock
  \item 1 tbsp fish sauce
  \item 4 tsp brown sugar
  \item Juice of $\tfrac{1}{2}$ lime
  \item \textbf{To serve:} Steamed jasmine rice, fresh chilli (sliced), lime wedges, fresh basil or Thai basil
\end{ingredients}

\begin{method}
  \item Heat sesame oil in a frying pan over medium heat. Add chicken and cook 5--10 minutes until browned.
  \item Stir in ginger and green curry paste, cooking 1--2 minutes until fragrant.
  \item Pour in coconut cream. Add bamboo shoots and capsicum, cook 5 minutes.
  \item Add chicken stock to the pan.
  \item Stir in fish sauce, brown sugar, and lime juice. Simmer 5--10 minutes until chicken is cooked through and capsicum softened.
  \item Taste and adjust flavours with extra fish sauce or sugar if needed.
  \item Serve with steamed rice, chilli, basil, and lime wedges.
\end{method}

\textbf{Source:} \href{https://vjcooks.com/chicken-thai-green-curry/#mv-creation-759-jtr}{VJ Cooks - Thai Green Curry}

\begin{notes}
\begin{itemize}
  \item \textbf{Heat level:} Green curry paste is medium heat. For milder curry, substitute yellow curry paste.
  \item \textbf{Vegetables:} Add eggplant, green beans, courgette, or carrots for extra veggies.
  \item \textbf{Protein swaps:} Replace chicken with prawns, beef, or tofu.
\end{itemize}
\end{notes}
\end{recipebox}

\begin{recipebox}{Classic Spaghetti Bolognese}{4 servings}{Prep 5 min, Cook 25 min (Total 30 min)}
\begin{ingredients}
  \item 1 tbsp olive oil
  \item 1 onion, finely diced
  \item 2 carrots, finely diced
  \item 2 celery sticks, finely diced
  \item 1 tbsp crushed garlic
  \item 500 g beef mince
  \item 3 tbsp tomato paste
  \item 1 tbsp Italian herbs
  \item 1 can (400 g) crushed tomatoes
  \item $\tfrac{1}{2}$ cup Beef Stock
  \item \textbf{To serve:} 500 g spaghetti (cooked), parmesan
\end{ingredients}

\begin{method}
  \item Heat olive oil in a large frying pan. Add onion, carrot, and celery. Cook, stirring, for about 5 minutes over medium heat.
  \item Stir in garlic. Add beef mince, breaking up with a wooden spoon. Cook 5 minutes until browned. Stir in tomato paste and herbs.
  \item Add Beef stock to pan with crushed tomatoes.
  \item Simmer 10--15 minutes on low, stirring occasionally, until liquid reduces and sauce thickens. Season with salt and pepper.
  \item Serve over spaghetti, topped with grated parmesan.
\end{method}

\textbf{Source:} \href{https://vjcooks.com/classic-spaghetti-bolognese/#mv-creation-346-jtr}{VJ Cooks – Classic Spaghetti Bolognese}

\begin{notes}
\begin{itemize}
  \item Freezes well. Double recipe to save one meal for later.
\end{itemize}
\end{notes}

\end{recipebox}

\begin{recipebox}{Loaded Beef Nachos}{3--4 servings}{Prep 10 min, Cook 20 min}
\begin{ingredients}
  \item 1 tbsp olive oil
  \item 1 onion, finely diced
  \item 2 cloves garlic, crushed
  \item 400 g beef mince
  \item 1 tbsp smoked paprika
  \item 1 tbsp ground cumin
  \item 1 tbsp tomato paste
  \item 1 tsp salt
  \item 1 can (400 g) crushed tomatoes
  \item 1 can (425 g) red kidney beans, rinsed and drained
  \item \textbf{To serve:}
  \begin{itemize}
    \item 250 g corn chips
    \item 100 g mozzarella, grated
    \item 1 avocado, diced
    \item 2 tomatoes, diced
    \item 125 g sour cream
    \item Fresh coriander (optional)
  \end{itemize}
\end{ingredients}

\begin{method}
  \item Heat olive oil in a large frying pan over medium heat. Add onion and sauté 5 minutes. Add garlic and mince, breaking up with a spatula. Cook until browned.
  \item Stir in paprika, cumin, tomato paste, and salt. Add crushed tomatoes and beans. Simmer on low 5 minutes.
  \item Preheat oven to 200\degree C fan grill.
  \item Remove mince mixture to a bowl. In the same pan (or baking dish), arrange corn chips, then spread mince evenly on top. Scatter with mozzarella.
  \item Grill 3--5 minutes until cheese melts.
  \item Top with avocado, tomato, sour cream, and coriander. Serve hot.
\end{method}

\textbf{Source:} \href{https://vjcooks.com/loaded-beef-nachos/#mv-creation-578-jtr}{VJ Cooks – Nachos}

\begin{notes}
\begin{itemize}
  \item Makes a big batch of mince — leftovers can be stored in the fridge or freezer for later.
\end{itemize}
\end{notes}
\end{recipebox}

\begin{recipebox}{Crunchy Crumbed Chicken Drumsticks}{4 servings}{Prep 15 min, Cook 45 min}
\begin{ingredients}
  \item 8 chicken drumsticks
  \item $\tfrac{1}{2}$ cup plain flour
  \item 3 large eggs, beaten
  \item 1 cup panko breadcrumbs
  \item Canola oil spray (or other oil)
  \item \textbf{Seasoning:}
  \begin{itemize}
    \item 1 tbsp dried oregano
    \item 1 tbsp dried basil
    \item 1 tsp garlic powder
    \item 1$\tfrac{1}{2}$ tsp smoked paprika
    \item 1$\tfrac{1}{2}$ tsp cayenne pepper or pure chilli powder (omit for kid-friendly)
    \item 1$\tfrac{1}{2}$ tsp cooking salt
    \item 2 tsp black pepper (reduce to $\tfrac{1}{2}$ tsp for kid-friendly)
  \end{itemize}
\end{ingredients}

\begin{method}
  \item Preheat oven to 200\degree C (180\degree C fan). Place a rack on a foil-lined tray and spray with oil.
  \item Prepare 3 bowls: flour, eggs, and panko.
  \item Mix seasoning in a small bowl. Stir 1 tbsp into flour, and the rest into panko.
  \item Pat drumsticks dry. Coat each in flour (shake off excess), then egg (let drip), then panko, pressing to adhere all over (see Note).
  \item Place on rack and spray generously with oil. Repeat with all drumsticks.
  \item Bake 25 min. Turn, spray again, and bake another 20 min until golden brown and crunchy.
  \item Serve hot and enjoy the crunch!
\end{method}

\textbf{Source:} \href{https://www.recipetineats.com/crunchy-crumbed-chicken-drumsticks/}{Crunchy Crumbed Chicken Drumsticks}

\begin{notes}
\begin{itemize}
  \item \textbf{Panko:} Japanese breadcrumbs for maximum crunch. Regular breadcrumbs may be substituted.
  \item \textbf{Mess-free crumbing:} Use one hand for flour/egg, the other to sprinkle and press panko.
  \item \textbf{Quick sauces:} Honey mustard (mayo, Dijon, honey, lemon), Pink sauce (sour cream + ketchup), or ketchup/BBQ sauce.
  \item Leftovers keep 3 days in fridge. Re-crisp in oven.
\end{itemize}
\end{notes}
\end{recipebox}

\begin{recipebox}{Harry’s Favourite Chocolate Cake}{10--12}{1 hr 15 min}
  \begin{ingredients}
    \item 1\nicefrac{2}{3} cups flour
    \item 1\nicefrac{1}{2} tsp baking soda
    \item 1\nicefrac{1}{2} cups sugar
    \item \nicefrac{2}{3} cup cocoa powder
    \item 1 tsp salt
    \item 1\nicefrac{1}{2} cups trim milk
    \item 100 g butter, melted (optional)
    \item 2 eggs
    \item 1 tsp vanilla essence
  \end{ingredients}

  \subsection*{Filling/Icing}
  \begin{itemize}[leftmargin=2em]
    \item 100 g butter
    \item 250 g icing sugar
    \item \nicefrac{1}{3} cup cocoa
    \item Milk, as needed
    \item 1 tsp vanilla essence
  \end{itemize}

  \begin{method}
    \item Preheat oven to $180^\circ$ C. Grease a springform tin (20-25 cm) and line with baking paper. (The mix is very liquid, so make sure it doesn’t leak.)
    \item Put all cake ingredients into a food processor and mix until combined.
    \item Pour into prepared tin and bake at least 50 minutes, or until a skewer comes out clean.
    \item To make icing: Place butter and dry ingredients in the food processor. Mix to combine.
    \item Add a dash of milk, mix again, then add vanilla.
    \item Cut cake in half. Spread filling and/or jam/berry purée in the centre. Replace the top and ice with the rest of the mixture.
  \end{method}
\end{recipebox}
