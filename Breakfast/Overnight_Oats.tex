\begin{recipebox}{Overnight Oats}{1 jar (make 5 for the week)}{5 min (plus overnight in the fridge)}
\textit{Makes 5 jars --- prep all at once on Monday evening.}

\begin{ingredients}
  \item[] \textit{Per jar:}
  \item \textonehalf{} cup rolled oats
  \item \textonehalf{} cup Greek yoghurt (full fat)
  \item 1 cup milk (or almond/oat milk)
  \item 1 scoop vanilla protein powder
  \item 1 tbsp chia seeds
  \item \textbf{Toppings (add in the morning):} banana, mixed berries, honey, granola
\end{ingredients}

\begin{method}
  \item Add oats, Greek yoghurt, milk, protein powder, and chia seeds to a jar or container. Stir well.
  \item Seal and refrigerate overnight (or at least 4 hours).
  \item In the morning, give it a stir and add your toppings.
  \item Grab and go --- no heating required.
\end{method}

\begin{notelist}
  \item \textbf{Tip:} Keeps in the fridge for up to 5 days. Prep all 5 jars on Sunday or Monday to cover the whole week.
\end{notelist}
\end{recipebox}
