\begin{recipebox}{Cottage Cheese \& Tomato Toast}{1}{--}
\begin{ingredients}
  \item 2 slices wholegrain bread, toasted
  \item \textonehalf{} cup cottage cheese
  \item 1 tomato, sliced
  \item Salt, pepper, dried herbs or chilli flakes
\end{ingredients}

\begin{method}
  \item Toast bread until golden.
  \item Spread cottage cheese generously on each slice.
  \item Top with sliced tomato. Season with salt, pepper, and herbs.
\end{method}

\begin{notelist}
  \item Approx. \$0.90/serve, 240 kcal, 22\,g protein per serve.
  \item \textbf{Tip:} Cottage cheese is one of the cheapest high-protein foods available --- around \$4--5 for a 500\,g tub at Pak'nSave. Also great with cucumber or smashed avocado on top.
\end{notelist}
\end{recipebox}
