\begin{recipebox}{Peanut Butter \& Banana Rice Cakes}{1}{--}
\begin{ingredients}
  \item 3 plain rice cakes
  \item 2 tbsp natural peanut butter
  \item 1 banana, sliced
  \item Pinch of cinnamon
\end{ingredients}

\begin{method}
  \item Spread peanut butter over each rice cake.
  \item Top with banana slices and a pinch of cinnamon.
\end{method}

\begin{notelist}
  \item Approx. \$0.80/serve, 250 kcal, 8\,g protein per serve.
  \item \textbf{Tip:} Lower protein than other snacks but high in healthy fats and carbs --- great as a pre-walk or post-exercise snack for energy. Add a scoop of protein powder to a glass of milk alongside it to boost protein.
\end{notelist}
\end{recipebox}
