\begin{recipebox}{Greek Yoghurt \& Berry Bowl}{1}{2 min}
\begin{ingredients}
  \item 200\,g full-fat Greek yoghurt
  \item \textonehalf{} cup mixed berries (fresh or frozen/thawed)
  \item 1 tbsp honey
  \item 1 tbsp granola or crushed almonds
\end{ingredients}

\begin{method}
  \item Spoon Greek yoghurt into a bowl.
  \item Top with berries, a drizzle of honey, and granola or almonds.
  \item Eat immediately.
\end{method}

\begin{notelist}
  \item \textbf{Tip:} Buy the 1\,kg tub of Anchor or Cyclops Greek yoghurt at Pak'nSave --- it's significantly cheaper per gram than individual pottles.
\end{notelist}
\end{recipebox}
