\begin{recipebox}{Greek Yogurt \& Berry Bowl}{1}{--}
\begin{ingredients}
  \item 200g full-fat Greek yogurt
  \item \textonehalf{} cup mixed berries (fresh or frozen/thawed)
  \item 1 tbsp honey
  \item 1 tbsp granola or crushed almonds
\end{ingredients}

\begin{method}
  \item Spoon Greek yogurt into a bowl.
  \item Top with berries, a drizzle of honey, and granola or almonds.
  \item Eat immediately.
\end{method}

\begin{notelist}
  \item Approx. \$1.20/serve, 220 kcal, 20g protein per serve.
  \item \textbf{Tip:} Buy the 1kg tub of Anchor or Cyclops Greek yogurt at Pak'nSave --- it's significantly cheaper per gram than individual pottles.
\end{notelist}
\end{recipebox}
