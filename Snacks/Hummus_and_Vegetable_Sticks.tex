\begin{recipebox}{Hummus \& Vegetable Sticks}{1}{--}
\begin{ingredients}
  \item \textonehalf{} cup hummus (store-bought or homemade)
  \item 2 carrots, cut into sticks
  \item 2 stalks celery, cut into sticks
  \item \textonehalf{} capsicum, cut into strips
  \item \textonehalf{} cucumber, cut into sticks
\end{ingredients}

\begin{method}
  \item Wash and cut all vegetables into dipping sticks.
  \item Portion hummus into a small container.
  \item Dip and enjoy.
\end{method}

%  PROTEIN SHAKE RECIPES

\begin{notelist}
  \item Approx. \$1.10/serve, 200 kcal, 9\,g protein per serve.
  \item \textbf{Tip:} Prep a container of cut vegetables at the start of the week and keep in the fridge --- it removes all friction from choosing this as a snack. Bluebird hummus is cheap and widely available at Pak'nSave.
\end{notelist}
\end{recipebox}
